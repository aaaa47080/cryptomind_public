SHELL=/bin/bash
PATH=/usr/local/bin:/usr/bin:/bin
# 每天 UTC 00:30 重算所有使用者的 trust_score。
# 先 source /etc/environment 取得 DATABASE_URL / REDIS_URL 等。
# 輸出導到 PID 1 的 stdout，讓 Zeabur Runtime Logs 看得到。
30 0 * * * . /etc/environment; cd /app && /usr/local/bin/python /app/scripts/cron_recompute_trust.py >> /proc/1/fd/1 2>&1
# 每 10 分鐘輪詢錢包監測（警示推送）。interval 由 WALLET_MONITOR_INTERVAL_MINUTES 控制。
*/10 * * * * . /etc/environment; cd /app && /usr/local/bin/python /app/scripts/cron_wallet_monitor.py >> /proc/1/fd/1 2>&1
